function uploadData() { const name = document.getElementById('student-name').value; const fileInput = document.getElementById('photo'); const status = document.getElementById('status'); const btn = document.getElementById('submit-btn'); const resetBtn = document.getElementById('reset-btn'); if (!name || fileInput.files.length === 0) { alert("이름과 사진을 확인해주세요."); return; } btn.disabled = true; status.innerHTML = "🚀 고해상도 최적화 및 전송 중... (조금만 기다려주세요)"; const file = fileInput.files[0]; const reader = new FileReader(); reader.onload = function(e) { const img = new Image(); img.src = e.target.result; img.onload = function() { const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d'); // QHD급(2560px) 해상도로 리사이징하여 용량을 1/10로 줄임 const TARGET_WIDTH = 2560; let width = img.width; let height = img.height; if (width > TARGET_WIDTH) { height = height * (TARGET_WIDTH / width); width = TARGET_WIDTH; } canvas.width = width; canvas.height = height; ctx.drawImage(img, 0, 0, width, height); // 80% 화질로 압축 (용량은 확 줄고, AI 가독성은 유지) const base64 = canvas.toDataURL('image/jpeg', 0.8).split(',')[1]; const data = { name: name, base64: base64 }; fetch(WEB_APP_URL, { method: "POST", body: JSON.stringify(data) }) .then(response => response.text()) .then(text => { status.innerHTML = "✅ 제출 및 채점 완료!"; btn.style.display = 'none'; resetBtn.style.display = 'block'; alert("성공적으로 제출되었습니다!"); }) .catch(err => { status.innerText = "❌ 전송 오류. 다시 시도해주세요."; btn.disabled = false; }); }; }; reader.readAsDataURL(file); }
